```js
// components/PaneContext.jsx
import React, {createContext, useContext, useEffect, useMemo, useState} from 'react';

const LS_KEY = 'pdfAnnotator:paneState:v1';
const DEFAULT = {
  sizes: { left: 260, middle: 1, right: 320 }, // px for side panes, flex for middle
  visibility: { left: true, middle: true, right: true },
  minimap: { enabled: false },
};

const PaneContext = createContext(null);
export const usePane = () => useContext(PaneContext);

export default function PaneProvider({children}) {
  const [state, setState] = useState(() => {
    try {
      const raw = localStorage.getItem(LS_KEY);
      if (!raw) return DEFAULT;
      const parsed = JSON.parse(raw);
      // Minimal migration/validation
      return {
        sizes: { ...DEFAULT.sizes, ...(parsed.sizes||{}) },
        visibility: { ...DEFAULT.visibility, ...(parsed.visibility||{}) },
        minimap: { ...DEFAULT.minimap, ...(parsed.minimap||{}) },
      };
    } catch {
      return DEFAULT;
    }
  });

  useEffect(() => {
    try { localStorage.setItem(LS_KEY, JSON.stringify(state)); } catch {}
  }, [state]);

  const api = useMemo(() => ({
    state,
    setSize: (pane, px) => setState(s => ({...s, sizes: {...s.sizes, [pane]: Math.max(160, px)}})),
    setVisibility: (pane, v) => setState(s => ({...s, visibility: {...s.visibility, [pane]: v}})),
    toggle: pane => setState(s => ({...s, visibility: {...s.visibility, [pane]: !s.visibility[pane]}})),
    reset: () => setState(DEFAULT),
  }), [state]);

  return <PaneContext.Provider value={api}>{children}</PaneContext.Provider>;
}
```

```js
// components/AnnotationContext.jsx
import React, {createContext, useContext, useMemo, useState} from 'react';

const AnnotationContext = createContext(null);
export const useAnno = () => useContext(AnnotationContext);

export default function AnnotationProvider({children}) {
  const [docMeta, setDocMeta] = useState({ pdfUrl: '', pageCount: 0, currentPage: 1 });
  const [tool, setTool] = useState('box'); // 'box' | 'pan' | 'select'
  const [annos, setAnnos] = useState({}); // page -> [{id, x,y,w,h, label, notes}]
  const [activeId, setActiveId] = useState(null);

  const api = useMemo(() => ({
    docMeta, setDocMeta, tool, setTool, activeId, setActiveId,
    annos,
    listForPage: (page) => annos[page] || [],
    upsert: (page, a) => setAnnos(prev => {
      const list = prev[page] || [];
      const idx = list.findIndex(x => x.id === a.id);
      const next = idx >= 0 ? [...list.slice(0, idx), a, ...list.slice(idx+1)] : [...list, a];
      return {...prev, [page]: next};
    }),
    remove: (page, id) => setAnnos(prev => ({...prev, [page]: (prev[page]||[]).filter(a => a.id !== id)})),
    replaceAllForPage: (page, nextList) => setAnnos(prev => ({...prev, [page]: nextList})),
  }), [annos, activeId, docMeta, tool]);

  return <AnnotationContext.Provider value={api}>{children}</AnnotationContext.Provider>;
}
```

```js
import React, {useRef} from 'react';
import {usePane} from './PaneContext';

export default function PaneContainer({left, middle, right}) {
  const {state, setSize, toggle} = usePane();
  const dragRef = useRef(null);

  const startDrag = (which, e) => {
    const startX = e.clientX;
    const startW = which === 'left' ? state.sizes.left : state.sizes.right;
    const onMove = (ev) => {
      const dx = ev.clientX - startX;
      const newSize = which === 'left' ? startW + dx : startW - dx;
      setSize(which, newSize);
    };
    const onUp = () => {
      window.removeEventListener('mousemove', onMove);
      window.removeEventListener('mouseup', onUp);
    };
    window.addEventListener('mousemove', onMove);
    window.addEventListener('mouseup', onUp);
  };

  return (
    <div className={styles.container}>
      {state.visibility.left && (
        <div className={styles.left} style={{width: state.sizes.left}}>
          {left}
        </div>
      )}
      <div
        className={styles.gutter}
        onMouseDown={(e) => startDrag('left', e)}
        title="Drag to resize (Left)"
      />
      {state.visibility.middle && (
        <div className={styles.middle} style={{flex: state.sizes.middle}}>
          {middle}
        </div>
      )}
      <div
        className={styles.gutter}
        onMouseDown={(e) => startDrag('right', e)}
        title="Drag to resize (Right)"
```

```js
// components/MiddlePane.jsx
import React, {useEffect, useRef, useState} from 'react';
import styles from '../styles/MiddlePane.module.css';
import {useAnno} from './AnnotationContext';
import {useToast} from './Toast';
import * as pdfjsLib from 'pdfjs-dist';
import 'pdfjs-dist/build/pdf.worker.entry';

pdfjsLib.GlobalWorkerOptions.workerSrc = undefined; // using bundler entry

function uuid() { return Math.random().toString(36).slice(2, 10); }

export default function MiddlePane() {
  const canvasRef = useRef(null);
  const overlayRef = useRef(null);
  const {docMeta, setDocMeta, tool, setTool, annos, listForPage, upsert, setActiveId, activeId} = useAnno();
  const toast = useToast();
  const [pdfDoc, setPdfDoc] = useState(null);
  const scale = 1.25;

  // Load PDF
  useEffect(() => {
    (async () => {
      if (!docMeta.pdfUrl) return;
      try {
        const doc = await pdfjsLib.getDocument(docMeta.pdfUrl).promise;
        setPdfDoc(doc);
        setDocMeta(m => ({...m, pageCount: doc.numPages, currentPage: 1}));
      } catch (e) {
        toast.error('Failed to load PDF.');
      }
    })();
  }, [docMeta.pdfUrl]);

  // Render current page
  useEffect(() => {
    (async () => {
      if (!pdfDoc || !docMeta.currentPage) return;
      const page = await pdfDoc.getPage(docMeta.currentPage);
      const viewport = page.getViewport({scale});
      const canvas = canvasRef.current;
      const ctx = canvas.getContext('2d');
      canvas.width = viewport.width; canvas.height = viewport.height;
      overlayRef.current.setAttribute('viewBox', `0 0 ${viewport.width} ${viewport.height}`);
      await page.render({canvasContext: ctx, viewport}).promise;
    })().catch(()=>toast.error('Failed to render page.'));
  }, [pdfDoc, docMeta.currentPage]);

  // Draw rectangles
  const [draft, setDraft] = useState(null); // {x,y,w,h}
  const onOverlayDown = (e) => {
    if (tool !== 'box') return;
    const pt = pointFromEvent(e);
    setDraft({x: pt.x, y: pt.y, w: 0, h: 0});
  };
  const onOverlayMove = (e) => {
    if (!draft || tool !== 'box') return;
    const pt = pointFromEvent(e);
    setDraft(d => ({...d, w: pt.x - d.x, h: pt.y - d.y}));
  };
  const onOverlayUp = () => {
    if (!draft) return;
    const norm = normalizeRect(draft);
    if (norm.w > 4 && norm.h > 4) {
      const id = uuid();
      upsert(docMeta.currentPage, {id, ...norm, label: '', notes: ''});
      setActiveId(id);
    }
    setDraft(null);
  };

  const pointFromEvent = (e) => {
    const svg = overlayRef.current;
    const rect = svg.getBoundingClientRect();
    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
  };
  const normalizeRect = (r) => {
    const w = Math.abs(r.w), h = Math.abs(r.h);
    const x = r.w < 0 ? r.x - w : r.x;
    const y = r.h < 0 ? r.y - h : r.y;
    return {x,y,w,h};
  };

  const selectAnno = (id) => setActiveId(id);

  return (
    <div className={styles.pane} onKeyDown={(e)=>{ if (e.key==='v') setTool('box'); if (e.key==='p') setTool('pan'); }}>
      <div className={styles.paneHeader}>
        <div className={styles.controls}>
          <span className={styles.badge}>Tool: {tool}</span>
        </div>
        <h3>PDF Viewer</h3>
      </div>

      <div className={styles.viewer}>
        <canvas ref={canvasRef} className={styles.canvas} />
        <svg
          ref={overlayRef}
          className={styles.overlay}
          onMouseDown={onOverlayDown}
          onMouseMove={onOverlayMove}
          onMouseUp={onOverlayUp}
        >
          {listForPage(docMeta.currentPage).map(a => (
            <g key={a.id} onClick={()=>selectAnno(a.id)}>
              <rect x={a.x} y={a.y} width={a.w} height={a.h}
                className={a.id===activeId ? styles.rectActive : styles.rect}/>
            </g>
          ))}
          {draft && (
            <rect x={normalizeRect(draft).x} y={normalizeRect(draft).y}
                  width={normalizeRect(draft).w} height={normalizeRect(draft).h}
                  className={styles.rectDraft}/>
          )}
        </svg>
      </div>
    </div>
  );
}
```

```js
import React, {useEffect} from 'react';
import {usePane} from './PaneContext';
import {useAnno} from './AnnotationContext';

export default function KeyboardShortcuts() {
  const {toggle} = usePane();
  const {setTool} = useAnno();

  useEffect(() => {
    const onKey = (e) => {
      const mod = e.metaKey || e.ctrlKey;
      if (mod && e.key.toLowerCase()==='b') { e.preventDefault(); toggle('left'); }    // VS Code explorer
      if (mod && e.shiftKey && e.key.toLowerCase()==='e') { e.preventDefault(); toggle('right'); }
      if (!mod && e.key.toLowerCase()==='v') { setTool('box'); }
      if (!mod && e.key.toLowerCase()==='p') { setTool('pan'); }
      if (!mod && e.key.toLowerCase()==='s') { setTool('select'); }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [toggle, setTool]);

  return null;
}
```

````md
# PDF Annotator (VS Code–style)

A responsive, three-pane PDF annotation prototype focused on speed, clarity, and modularity. Uses React + pdf.js, with pane state persisted in `localStorage`.

## Panes
- **Left (Explorer):** Load PDF, navigate pages, export utilities.
- **Middle (Viewer):** pdf.js canvas + SVG overlay. Tools: Box select (V), Pan (P), Select (S).
- **Right (Inspector):** Edit label & notes for the selected region; shows geometry.

## LocalStorage Schema
**Key:** `pdfAnnotator:paneState:v1`

**Shape:**
```json
{
  "sizes": { "left": 260, "middle": 1, "right": 320 },
  "visibility": { "left": true, "middle": true, "right": true },
  "minimap": { "enabled": false }
}
````

* `sizes.left`/`sizes.right`: number (px, min 160).
* `sizes.middle`: number (flex grow weight).
* `visibility.*`: boolean.
* `minimap.enabled`: boolean; reserved for future UI.

**Notes:** On load, values are validated and merged with defaults. If parsing fails, defaults are used.

## Interaction Guide (Data Flow)

1. **Load PDF:** Explorer sets `docMeta.pdfUrl`; Middle loads via pdf.js and updates `pageCount`.
2. **Navigate:** Explorer sets `docMeta.currentPage`; Middle renders canvas and resets overlay.
3. **Annotate:** In Viewer, press **V** (box tool). Drag to create a rectangle; it’s added to `annos[currentPage]` and becomes active.
4. **Edit:** Selecting a rectangle sets `activeId`; Inspector binds `label`/`notes` to that record and updates via `upsert`.
5. **HUD (Optional):** When an annotation is selected, a draggable HUD appears near it for quick edits.
6. **Persist:** Pane visibility/sizes are persisted automatically to `localStorage`.

## Keyboard Shortcuts

* **⌘/Ctrl+B:** Toggle Explorer (left pane)
* **⌘/Ctrl+Shift+E:** Toggle Inspector (right pane)
* **V:** Box tool
* **P:** Pan tool
* **S:** Select tool

## Error Handling

* **Loading/Rendering Failures:** Non-blocking toasts (`ToastProvider`) with contextual messages.
* **Validation (Inspector):** Basic required checks for label/notes can be added inline (e.g., outline invalid fields).
* **Destructive Actions:** Deletions require `confirm()`; cancellation leaves data unchanged.
* **Resilience:** Pane schema read is guarded; corrupted entries fall back to defaults.

## Performance Notes

* Canvas renders only on page change; annotation overlay uses lightweight SVG elements.
* Minimal state updates with page-scoped lists; object URLs used for local PDFs.
* CSS-only HUD/overlay; no expensive layout thrash.

## Development

* Components are modular and documented.
* Tests (Jest + RTL) cover pane toggling/resizing/persistence and basic annotation flow.

```js
// __tests__/PaneContainer.panes.test.jsx
import React from 'react';
import {render, fireEvent} from '@testing-library/react';
import PaneProvider from '../components/PaneContext';
import PaneContainer from '../components/PaneContainer';

function setup() {
  localStorage.clear();
  return render(
    <PaneProvider>
      <PaneContainer left={<div>left</div>} middle={<div>mid</div>} right={<div>right</div>} />
    </PaneProvider>
  );
}

test('toggles left pane visibility and persists', () => {
  const {getByLabelText, container} = setup();
  // Initially visible
  expect(container.textContent).toContain('left');
  fireEvent.click(getByLabelText('Toggle Explorer'));
  // Hidden after toggle
  expect(container.textContent).not.toContain('left');
  // Persistence check
  const saved = JSON.parse(localStorage.getItem('pdfAnnotator:paneState:v1'));
  expect(saved.visibility.left).toBe(false);
});

test('drag left gutter resizes left pane (min 160)', () => {
  const {getByTitle, container} = setup();
  const gutter = getByTitle('Drag to resize (Left)');
  // Simulate drag +50 px
  fireEvent.mouseDown(gutter, {clientX: 200});
  fireEvent.mouseMove(window, {clientX: 250});
  fireEvent.mouseUp(window);
  const saved = JSON.parse(localStorage.getItem('pdfAnnotator:paneState:v1'));
  expect(saved.sizes.left).toBeGreaterThanOrEqual(160);
});
```

```js
import React from 'react';
import {render, fireEvent} from '@testing-library/react';
import AnnotationProvider from '../components/AnnotationContext';
import MiddlePane from '../components/MiddlePane';
import ToastProvider from '../components/Toast';

// Mock pdfjs and canvas to avoid real rendering
jest.mock('pdfjs-dist', () => ({
  getDocument: () => ({ promise: Promise.resolve({
    numPages: 1,
    getPage: async () => ({
      getViewport: ({scale}) => ({ width: 600, height: 800 }),
      render: () => ({ promise: Promise.resolve() })
    })
  })}),
  GlobalWorkerOptions: { workerSrc: undefined }
}));

HTMLCanvasElement.prototype.getContext = () => ({ drawImage: ()=>{}, clearRect: ()=>{} });

function Wrapper({children}) {
  return (
    <ToastProvider>
      <AnnotationProvider>{children}</AnnotationProvider>
    </ToastProvider>
  );
}

test('creates a rectangle annotation on drag', async () => {
  const {findByText, container} = render(<Wrapper><MiddlePane/></Wrapper>);
  // Inject a fake pdf url to trigger render
  // Directly dispatch to context by setting attribute is complex here;
  // Instead, mount then simulate overlay drag: we know SVG will exist
  const overlay = await new Promise(resolve => {
    const iv = setInterval(()=>{
      const el = container.querySelector('svg');
      if (el) { clearInterval(iv); resolve(el); }
    }, 10);
  });
  // Tool defaults to 'box'; simulate drag
  fireEvent.mouseDown(overlay, {clientX: 100, clientY: 100});
  fireEvent.mouseMove(overlay, {clientX: 200, clientY: 220});
  fireEvent.mouseUp(overlay);
  // Expect a rect draft gone and a final rect exists
  const rects = container.querySelectorAll('rect');
  expect(rects.length).toBeGreaterThan(0);
});
```
