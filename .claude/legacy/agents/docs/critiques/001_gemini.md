### **Task 1 Artifact: `create_header_analysis_tasks.py`**

This script handles the discovery phase. It finds the suspicious nodes and prepares the work for the `section-fixer` agent.

```python
#!/usr/bin/env python3
"""
Scans a large marker-pdf JSON file to identify nodes with a 'suspicious_header'
flag set to true. It then generates a structured JSON task list, printing it
to standard output.

Each task in the list contains the suspicious node, its surrounding nodes for
context, and a directive for a downstream agent.
"""

import json
import subprocess
import sys
from pathlib import Path

# --- Configuration ---
CONTEXT_WINDOW_SIZE = 2 # Number of nodes before and after to include
JSON_FILE_PATH = "large_document.json" # The target file

def get_suspicious_header_indices(json_file_path: str) -> list[int]:
    """Uses 'jq' to efficiently find the array indices of all suspicious headers."""
    print(f"[*] Running jq to find suspicious header indices in {json_file_path}...", file=sys.stderr)
    jq_command = "jq 'to_entries | .[] | select(.value.header_info?.suspicious_header? == true) | .key'"
    
    try:
        process = subprocess.run(
            f"cat {json_file_path} | {jq_command}",
            shell=True, check=True, capture_output=True, text=True
        )
        if not process.stdout.strip():
            return []
        indices = [int(i) for i in process.stdout.strip().split('\n')]
        print(f"[+] Found {len(indices)} suspicious candidates.", file=sys.stderr)
        return indices
    except subprocess.CalledProcessError as e:
        print(f"[!] Error running jq: {e.stderr}", file=sys.stderr)
        return []
    except FileNotFoundError:
        print("[!] Error: 'jq' command not found.", file=sys.stderr)
        return []

def create_analysis_tasks(all_blocks: list, indices: list) -> list[dict]:
    """Creates a list of tasks with context and a directive for the fixing agent."""
    print(f"[*] Creating {len(indices)} analysis tasks with context...", file=sys.stderr)
    task_list = []
    for i, original_index in enumerate(indices):
        start = max(0, original_index - CONTEXT_WINDOW_SIZE)
        end = min(len(all_blocks), original_index + CONTEXT_WINDOW_SIZE + 1)
        
        context_slice = all_blocks[start:end]
        candidate_index_in_slice = original_index - start
        
        task = {
            "task_id": f"header_fix_{i}",
            "original_document_index": original_index,
            "candidate_index_in_slice": candidate_index_in_slice,
            "context_slice": context_slice,
            "directive": (
                "Analyze the candidate node within the context slice. Determine if it is a "
                "TRUE_HEADER or if it is misidentified. If misidentified, suggest the correct "
                "node type (e.g., Text, Caption, ListItem). Your output should be a classification."
            )
        }
        task_list.append(task)
    return task_list

def main():
    """Main execution function."""
    json_path = Path(JSON_FILE_PATH)
    if not json_path.exists():
        print(f"[!] Error: The file {json_path} was not found.", file=sys.stderr)
        sys.exit(1)
        
    try:
        # Load the file once to create context slices. This is memory-intensive but
        # necessary for slicing and is acceptable on a high-RAM machine.
        with open(json_path, 'r', encoding='utf-8') as f:
            all_pdf_blocks = json.load(f)
            
        suspicious_indices = get_suspicious_header_indices(str(json_path))
        if not suspicious_indices:
            print(json.dumps([])) # Output empty list for no work
            sys.exit(0)
            
        task_list = create_analysis_tasks(all_pdf_blocks, suspicious_indices)
        
        # Print the entire task list as a single JSON string to stdout for the agent
        print(json.dumps(task_list))
        
    except Exception as e:
        print(f"[!] An error occurred: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
```

```python
def _analyze_tasks_and_build_jq_filter(tasks: List[Dict]) -> str:
    """
    Analyzes each task and builds a single jq filter string for all changes.
    
    In a real implementation, this function would contain calls to an LLM
    for each task to determine the classification. Here, we simulate it.
    """
    logger.info(f"Analyzing {len(tasks)} tasks to determine necessary changes...")
    jq_filters = []
    
    for task in tasks:
        # --- SIMULATED LLM REASONING ---
        # This is where you would call your reasoning agent (Claude/Ollama)
        # with the task's context_slice and directive.
        # For this example, we'll use simple heuristics.
        candidate_text = task['context_slice'][task['candidate_index_in_slice']]['text']
        
        # Heuristic: If it starts with a bullet or is long, it's not a header.
        if candidate_text.startswith(("-", "*")) or len(candidate_text.split()) > 10:
            classification = "Text" # Reclassify the node type
        # Heuristic: The example false positive from the user.
        elif "False Section Header" in candidate_text:
            classification = "Text"
        else:
            # Otherwise, assume the only issue was the suspicious flag.
            classification = "TRUE_HEADER"
        # --- END SIMULATION ---

        index = task['original_document_index']
        
        if classification == "TRUE_HEADER":
            # The node is a real header, just fix the suspicious flag.
            jq_filters.append(f".[{index}].header_info.suspicious_header = false")
        else:
            # The node is misidentified. Change its type AND fix the flag.
            jq_filters.append(
                f".[{index}].type = \"{classification}\" | "
                f".[{index}].header_info.suspicious_header = false"
            )
            
    if not jq_filters:
        return ""
        
    # Join all individual filters into a single, massive command
    return " | ".join(jq_filters)
```

Excellent question. This moves us from "how to build it" to "should we build it this way?"

Given the JSON list you provided and the context that a real-world file could be 10 times the size of Claude's context window, the answer is an emphatic **yes, this is a realistic and, in fact, an optimal approach.**

It is not overly complex; it is *appropriately* complex for the scale of the problem. A simpler approach would fail. Let's break down the pros and potential pitfalls of this agentic pipeline.

---

### The Pros: Why This Approach is Realistic and Powerful

1.  **Solves the Massive Context Window Problem (The Core Issue):**
    This is the most significant advantage. By refusing to load the entire multi-gigabyte file into a single prompt, you sidestep the context limit entirely. The `create_header_analysis_tasks.py` script acts as a "smart pre-processor." It reads the large file on a machine with sufficient RAM and carves out tiny, context-rich "slices" that are easily digestible for any LLM. The agent only ever sees a few KB of data at a time, even though it's operating on a GB-scale file.

2.  **Efficiency and Scalability (The `jq` Magic):**
    Your directive to use `jq` to apply the final changes is brilliant and critical.
    *   **Memory Safety:** `jq` streams the input file. It does not load the entire gigabyte file into its own memory. It reads, applies the filter to the relevant object, writes, and moves on. This is extremely memory-efficient.
    *   **I/O Efficiency:** By building a single, large `jq` command that applies all fixes in one pass (`jq '.[123].type="Text" | .[456].type="Text" | ...'`), you read the massive source file **once** and write the corrected version **once**. The alternative—running one `jq` command per fix—would involve reading and writing the entire gigabyte file thousands of times, which would be catastrophically slow.

3.  **Modularity and Maintainability:**
    The two-task structure makes the system far easier to manage, debug, and improve.
    *   **Discovery (`create_..._tasks.py`):** This script's only job is to find candidates. If your criteria for "suspicious" changes, you only modify this one script.
    *   **Reasoning (`section-fixer` agent):** This agent's only job is to analyze a small slice of context. If you want to improve its analytical abilities (e.g., switch from a simple heuristic to a more powerful LLM call, add more logic), you only change the `section-fixer` worker. The two parts are decoupled.

4.  **Robustness and Error Handling:**
    This workflow is more robust than a monolithic approach.
    *   **Atomic Updates:** The `... > temp_file && mv temp_file original_file` pattern is an atomic operation on most filesystems. This means the update either completes successfully or the original file is left untouched. You won't end up with a corrupted, half-written file if the script fails midway.
    *   **Knowledge Architect Integration:** Your framework is built for this. The `ToolJourneyTracker` will record the entire process. If the `jq` command fails, you have a record of exactly what command was attempted. If the analysis of one task fails, the agent can be programmed to skip it and continue with the rest, ensuring that one bad node doesn't halt the entire batch.

### The Potential Pitfalls: What to Watch Out For

While this approach is optimal, it's not without its challenges. These are not reasons to avoid it, but rather factors to be aware of during implementation.

1.  **Complexity of the `jq` "God Command":**
    The strategy hinges on building one massive `jq` filter string. While `jq` can handle very complex filters, operating systems have a limit on the maximum length of a command-line argument (`ARG_MAX`). On modern Linux systems, this is very large (often 2MB+), but it's not infinite. If you needed to correct, say, 500,000 nodes, the generated filter string *could* potentially hit this limit.
    *   **Mitigation:** For extreme cases, the `section-fixer` worker could be designed to "batch" its fixes. If it has 500,000 corrections to make, it could generate 10 `jq` commands that each apply 50,000 fixes, running them sequentially against the progressively updated file.

2.  **The Reasoning Bottleneck:**
    The slowest and most expensive part of this workflow is the analysis inside the `section-fixer` agent, where it must iterate through the task list. If you have 5,000 suspicious headers, this means 5,000 reasoning steps (potentially 5,000 LLM calls).
    *   **Mitigation:** This is where parallelism is key. The `section-fixer` agent should be designed to analyze the tasks concurrently using `asyncio` to make many LLM API calls in parallel. This turns a long sequential process into a much shorter, wide parallel one. This is also where a hybrid model (a fast local Ollama model for easy cases, escalating to Claude for hard ones) would provide massive speed and cost benefits.

3.  **Error Propagation and Debugging:**
    Because this is a multi-step pipeline, an error in an early stage can cause problems later. If the `create_..._tasks.py` script generates incorrect indices, the `jq` command will silently modify the wrong parts of the document.
    *   **Mitigation:** Rigorous testing and validation are crucial. The `working_usage()` functions in your worker templates are essential. Before running on a real gigabyte file, test the entire end-to-end flow on a small, representative sample (like the one you asked me to generate) to ensure the indices are correct and the `jq` filter is constructed as expected. Logging within the `section-fixer` is also vital to trace which decision led to which part of the `jq` command.

```python
#!/usr/bin/env python3
"""
A targeted discovery script that safely and efficiently identifies work items
(nodes with 'suspicious_header': true) from a large marker-pdf JSON file.

This script takes a file path as input, uses the 'jq' utility to find
candidate nodes, and outputs a stream of tasks as Newline Delimited JSON (NDJSON)
to standard output. This output can be piped directly to downstream agents.
"""

import argparse
import json
import logging
import subprocess
import sys
from pathlib import Path

# --- Configuration ---
# Set up structured logging for better observability in containerized environments.
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    stream=sys.stderr
)

def get_suspicious_header_indices(json_path: Path) -> list[int]:
    """
    Safely uses 'jq' to find the array indices of all suspicious headers.
    This function avoids shell injection by passing arguments as a list.
    """
    if not json_path.is_file():
        logging.error(f"Input path is not a file: {json_path}")
        return []

    jq_filter = 'to_entries | .[] | select(.value.header_info?.suspicious_header? == true) | .key'
    command = ["jq", jq_filter, str(json_path)]
    
    logging.info(f"Running command: {' '.join(command)}")

    try:
        # Use check=True to raise CalledProcessError on non-zero exit codes.
        # This prevents silently failing on malformed JSON or if jq is missing.
        process = subprocess.run(
            command,
            check=True,
            capture_output=True,
            text=True,
        )
        output = process.stdout.strip()
        if not output:
            logging.info("No suspicious headers found.")
            return []
            
        indices = [int(i) for i in output.split('\n')]
        logging.info(f"Found {len(indices)} suspicious header candidates.")
        return indices
        
    except FileNotFoundError:
        logging.critical("'jq' command not found. Please ensure it is installed and in your PATH.")
        sys.exit(1)
    except subprocess.CalledProcessError as e:
        logging.error(f"jq command failed with exit code {e.returncode}:")
        logging.error(f"Stderr: {e.stderr}")
        return []

def create_and_stream_tasks(all_blocks: list, indices: list, context_window: int):
    """
    Creates and streams tasks as NDJSON to stdout.
    This avoids building a giant list in memory and allows the consumer
    to process tasks as they are generated.
    """
    logging.info(f"Streaming {len(indices)} analysis tasks with context window {context_window}...")
    task_count = 0
    for i, original_index in enumerate(indices):
        start = max(0, original_index - context_window)
        end = min(len(all_blocks), original_index + context_window + 1)
        
        context_slice = all_blocks[start:end]
        candidate_index_in_slice = original_index - start
        
        task = {
            "task_id": f"header_fix_{i}",
            "original_document_index": original_index,
            "candidate_index_in_slice": candidate_index_in_slice,
            "context_slice": context_slice,
            "directive": (
                "Analyze the candidate node to determine if it is a TRUE_HEADER or misidentified. "
                "Suggest the correct node type (e.g., Text, Caption) if misidentified."
            )
        }
        # Print each task as a separate JSON line (NDJSON)
        print(json.dumps(task))
        task_count += 1
    logging.info(f"Successfully streamed {task_count} tasks.")

def main():
    """Main execution function with CLI argument parsing."""
    parser = argparse.ArgumentParser(
        description="Finds suspicious headers in a marker-pdf JSON file and generates tasks.",
        formatter_class=argparse.ArgumentDefaultsHelpFormatter
    )
    parser.add_argument("file", type=Path, help="Path to the large JSON file to process.")
    parser.add_argument(
        "--context", type=int, default=2,
        help="Number of nodes to include before and after the candidate for context."
    )
    args = parser.parse_args()

    # NOTE: This script loads the entire file into memory to create context slices.
    # This is a deliberate design choice for simplicity, assuming a high-RAM environment
    # as specified in the original problem. A fully streaming alternative would be
    # significantly more complex, requiring multiple passes over the file.
    try:
        logging.info(f"Loading {args.file} into memory to create context slices...")
        with args.file.open('r', encoding='utf-8') as f:
            all_pdf_blocks = json.load(f)
    except (json.JSONDecodeError, MemoryError) as e:
        logging.critical(f"Failed to load or parse JSON file: {e}")
        sys.exit(1)

    suspicious_indices = get_suspicious_header_indices(args.file)
    if suspicious_indices:
        create_and_stream_tasks(all_pdf_blocks, suspicious_indices, args.context)

if __name__ == "__main__":
    main()
```

```python
"""
Worker for the section-fixer agent.

This worker takes a list of analysis tasks for a marker-pdf JSON file,
determines the necessary corrections, and constructs batched `jq` commands
to apply all fixes efficiently and safely. It fully integrates with the
Knowledge Architect framework.
"""

import asyncio
import json
import logging
import shlex
import shutil
import subprocess
import sys
from abc import ABC, abstractmethod
from pathlib import Path
from typing import Dict, List, Any

from loguru import logger
from agents.workers.knowledge_architect_worker import (
    ToolJourneyTracker,
    create_solution_relationships,
    check_existing_solutions,
    extract_task_type
)

# --- Configuration ---
AGENT_NAME = "section-fixer"
MAX_OPS_PER_JQ_COMMAND = 500  # Batch size to avoid ARG_MAX limit
CONCURRENCY_LIMIT = 10        # Max parallel analysis tasks

# --- Pluggable Classifier System ---

class BaseClassifier(ABC):
    """Abstract base class for classifiers."""
    @abstractmethod
    async def classify(self, task: Dict[str, Any]) -> str:
        """Takes a task and returns a classification string (e.g., 'Text', 'TRUE_HEADER')."""
        pass

class HeuristicClassifier(BaseClassifier):
    """A simple, deterministic classifier based on heuristics."""
    async def classify(self, task: Dict[str, Any]) -> str:
        # This heuristic is for demonstration; it's flawed but deterministic.
        # A real system would use a more robust classifier.
        candidate_text = task['context_slice'][task['candidate_index_in_slice']]['text']
        if candidate_text.startswith(("-", "*", "–")) or len(candidate_text.split()) > 10:
            return "Text"
        elif "False Section Header" in candidate_text:
            return "Text"
        return "TRUE_HEADER"

class LLMClassifier(BaseClassifier):
    """Placeholder for a real LLM-based classifier."""
    async def classify(self, task: Dict[str, Any]) -> str:
        # In a real implementation, this would:
        # 1. Format a prompt with the context_slice and directive.
        # 2. Make an awaitable call to an LLM API (e.g., Claude, Ollama).
        # 3. Parse the LLM's response to get the classification.
        # 4. Handle rate limiting and errors.
        logger.warning("Using placeholder LLMClassifier. Falling back to HeuristicClassifier.")
        heuristic_classifier = HeuristicClassifier()
        return await heuristic_classifier.classify(task)

def get_classifier(classifier_type: str = "heuristic") -> BaseClassifier:
    """Factory function to get a classifier instance."""
    if classifier_type == "llm":
        return LLMClassifier()
    return HeuristicClassifier()

# --- Core Worker Logic ---

async def _analyze_tasks_and_build_jq_filters(tasks: List[Dict], classifier: BaseClassifier) -> List[str]:
    """
    Analyzes tasks concurrently and builds a list of individual jq filter strings.
    """
    logging.info(f"Analyzing {len(tasks)} tasks concurrently (limit: {CONCURRENCY_LIMIT})...")
    
    semaphore = asyncio.Semaphore(CONCURRENCY_LIMIT)
    
    async def classify_with_semaphore(task):
        async with semaphore:
            return await classifier.classify(task)

    classifications = await asyncio.gather(*(classify_with_semaphore(task) for task in tasks))
    
    jq_filters = []
    for task, classification in zip(tasks, classifications):
        index = task['original_document_index']
        if classification == "TRUE_HEADER":
            jq_filters.append(f".[{index}].header_info.suspicious_header = false")
        else:
            jq_filters.append(
                f".[{index}].type = \"{classification}\" | "
                f".[{index}].header_info.suspicious_header = false"
            )
    return jq_filters

def _execute_jq_fix_batch(json_path: Path, jq_filter_batch: str) -> Dict[str, Any]:
    """
    Executes a single batch of jq fixes safely.
    It creates a backup, uses safe subprocess calls, and performs an atomic move.
    """
    temp_path = json_path.with_suffix(".json.tmp")
    command = ["jq", jq_filter_batch, str(json_path)]
    
    try:
        with open(temp_path, "w", encoding="utf-8") as out_file:
            subprocess.run(command, stdout=out_file, check=True, text=True)
        # Atomic replacement of the original file
        temp_path.replace(json_path)
        return {"success": True}
    except subprocess.CalledProcessError as e:
        error_msg = f"jq command failed with exit code {e.returncode}: {e.stderr}"
        logging.error(error_msg)
        # Clean up temporary file on failure
        if temp_path.exists():
            temp_path.unlink()
        return {"success": False, "error": error_msg}

async def run_header_analysis_and_fix(task_list_json: str, file_path_str: str, dry_run: bool = False) -> Dict[str, Any]:
    """
    Primary function for the section-fixer worker.
    Orchestrates analysis, batched jq command generation, and safe execution.
    """
    file_path = Path(file_path_str)
    tasks = [json.loads(line) for line in task_list_json.strip().split('\n')]
    task_description = f"Correct document structure in {file_path.name} for {len(tasks)} candidates."
    
    journey = ToolJourneyTracker(extract_task_type(task_description), task_description)
    
    try:
        if not shutil.which("jq"):
            raise FileNotFoundError("jq command is not available in the system PATH.")
        if not file_path.exists():
            raise FileNotFoundError(f"Target file does not exist: {file_path}")

        classifier = get_classifier("llm") # Default to the smarter classifier
        
        step_idx = journey.add_step("analysis", "classify_nodes", {"task_count": len(tasks)})
        all_filters = await _analyze_tasks_and_build_jq_filters(tasks, classifier)
        journey.complete_step(step_idx, True, f"Generated {len(all_filters)} potential modifications.")

        if not all_filters:
            return {'success': True, 'data': "Analysis complete. No changes needed."}

        # Batch filters to avoid ARG_MAX
        filter_batches = [
            " | ".join(all_filters[i:i + MAX_OPS_PER_JQ_COMMAND])
            for i in range(0, len(all_filters), MAX_OPS_PER_JQ_COMMAND)
        ]

        if dry_run:
            logging.warning("DRY RUN enabled. Printing commands without executing.")
            for i, batch in enumerate(filter_batches):
                print(f"--- BATCH {i+1}/{len(filter_batches)} ---\njq '{batch}' '{file_path}'\n")
            return {'success': True, 'data': f"Dry run complete. Found {len(all_filters)} changes in {len(filter_batches)} batches."}

        # Create a single backup before any modifications
        backup_path = file_path.with_suffix(".json.bak")
        shutil.copy(file_path, backup_path)
        logging.info(f"Created backup at {backup_path}")

        step_idx = journey.add_step("jq_execution", "apply_fixes", {"batch_count": len(filter_batches)})
        for i, batch in enumerate(filter_batches):
            logging.info(f"Executing batch {i+1}/{len(filter_batches)}...")
            result = _execute_jq_fix_batch(file_path, batch)
            if not result['success']:
                raise RuntimeError(f"Batch {i+1} failed: {result['error']}")
        
        journey.complete_step(step_idx, True, f"Successfully applied {len(all_filters)} changes.")
        journey.finish_journey("success")
        journey.save_successful_journey()
        create_solution_relationships(problem=task_description, solution="Applied fixes via batched jq commands.", tool_journey=journey.journey)
        
        return {'success': True, 'data': f"Successfully applied {len(all_filters)} corrections."}

    except Exception as e:
        logging.critical(f"Header analysis and fix failed: {e}", exc_info=True)
        journey.finish_journey("failed", str(e))
        return {'success': False, 'error': str(e), 'metrics': {'tool_journey': journey.journey}}
```
